deleted; a repo-wide grep is the completeness check.

import json
from pathlib import Path

REPO = Path("/Users/trishasalas/Repos/Research/tmlr")
NB = REPO / "notebooks"


def load(name):
    return json.loads((NB / name).read_text())


def save(name, nb):
    with open(NB / name, "w") as f:
        json.dump(nb, f, indent=1, ensure_ascii=False)
        f.write("\n")


def cell_src(cell):
    s = cell["source"]
    return "".join(s) if isinstance(s, list) else s


def find_cell(nb, needle, name):
    """Locate by CONTENT — cell ids collide across these notebooks."""
    hits = [c for c in nb["cells"] if needle in cell_src(c)]
    assert len(hits) == 1, f"{name}: expected 1 cell containing {needle!r}, got {len(hits)}"
    return hits[0]


def patch_cell(cell, replacements, name):
    src = cell_src(cell)
    for old, new in replacements:
        n = src.count(old)
        assert n == 1, f"{name}: expected 1 occurrence of {old[:50]!r}, got {n}"
        src = src.replace(old, new)
    cell["source"] = src.splitlines(keepends=True)


ENTROPY_CALL_OLD = """create_manifest_file(
    PROJECT_ROOT, output_dir, model_name, model,
    "entropy", domain_counts, results_df, prompt_files,"""
ENTROPY_CALL_NEW = """write_entropy_manifest(
    PROJECT_ROOT, output_dir, model_name, model,
    domain_counts, results_df, prompt_files,"""

BINDING_CALL_OLD = """create_manifest_file(
    PROJECT_ROOT, output_dir, model_name, model,
    "binding", domain_counts, results_df, prompt_files,"""
BINDING_CALL_NEW = """write_binding_manifest(
    PROJECT_ROOT, output_dir, model_name, model,
    domain_counts, results_df, prompt_files,"""

JOBS = [
    ("binding-gpt2.ipynb", [
        ("from src.manifest import create_manifest_file",
         "from src.manifest import write_binding_manifest"),
        (BINDING_CALL_OLD, BINDING_CALL_NEW),
    ]),
    ("entropy-gpt2.ipynb", [
        ("from src.manifest import create_manifest_file",
         "from src.manifest import write_entropy_manifest"),
        (ENTROPY_CALL_OLD, ENTROPY_CALL_NEW),
    ]),
    ("entropy-pythia.ipynb", [
        ("from src.manifest import create_manifest_file",
         "from src.manifest import write_entropy_manifest"),
        (ENTROPY_CALL_OLD, ENTROPY_CALL_NEW),
    ]),
    ("entropy-olmo.ipynb", [
        ("from src.manifest import create_manifest_file",
         "from src.manifest import write_entropy_manifest"),
        (ENTROPY_CALL_OLD, ENTROPY_CALL_NEW),
    ]),
]

for name, replacements in JOBS:
    nb = load(name)
    cell = find_cell(nb, "create_manifest_file(", name)
    patch_cell(cell, replacements, name)
    save(name, nb)
    print(f"  patched {name}")

print("\n4 forward-pass notebooks patched")
